P: Napisz skrypt ~/bin/kto-na-porcie.sh, który dla podanego portu wypisuje proces, użytkownika i adres nasłuchu albo informuje, że port jest wolny.
H: awk z dokładnym dopasowaniem portu i ps.
S:
cat > ~/bin/kto-na-porcie.sh <<'EOF'
#!/bin/bash
p=$1
linia=$(sudo netstat -plnt | awk -v p=":$p\$" '$4 ~ p {print $4, $7; exit}')
[ -z "$linia" ] && { echo "port $p wolny"; exit 1; }
adres=${linia%% *}; prog=${linia#* }; pid=${prog%%/*}
echo "port $p: $adres, proces $prog, uzytkownik $(ps -o user= -p "$pid" 2>/dev/null)"
EOF
chmod +x ~/bin/kto-na-porcie.sh
~/bin/kto-na-porcie.sh 8080
~/bin/kto-na-porcie.sh 9999
