ssh anna@203.0.113.50 'sudo tee /usr/local/bin/wdroz-sklep.sh >/dev/null <<"EOF"
#!/bin/bash
set -euo pipefail
loguj() { echo "wdroz-sklep.sh: [$1] ${*:2}"; }
zakoncz() { loguj err "${*:2}"; exit "$1"; }
czekaj_na_gotowosc() {
    local port=$1 limit=$2 i=0
    while (( i < limit )); do
        curl -fs "127.0.0.1:$port/actuator/health/readiness" >/dev/null 2>&1 && return 0
        sleep 1; ((i++))
    done
    return 1
}
retencja() { ls -1dt /opt/sklep/releases/*/ | tail -n +"$(( $1 + 1 ))" | xargs -r rm -rf; }
main() {
    local wersja=$1 suma=$2 katalog=/opt/sklep/releases/$1 poprzednie
    [[ $wersja =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]] || zakoncz 2 "zla wersja: $wersja"
    install -d -m 755 "$katalog"
    cat > "$katalog/sklep.jar"
    echo "$suma  $katalog/sklep.jar" | sha256sum -c --quiet || { rm -rf "$katalog"; zakoncz 2 "zla suma kontrolna"; }
    poprzednie=$(readlink -f /opt/sklep/sklep.jar || true)
    ln -sfn "$katalog/sklep.jar" /opt/sklep/.sklep.jar.nowy && mv -T /opt/sklep/.sklep.jar.nowy /opt/sklep/sklep.jar
    systemctl restart sklep
    if czekaj_na_gotowosc 8080 90; then
        loguj info "wdrozono $wersja (wdrazal: ${SUDO_USER:-$USER})"; retencja 5
    else
        loguj err "brak gotowosci $wersja - wycofanie do ${poprzednie:-brak}"
        [[ -n $poprzednie ]] && ln -sfn "$poprzednie" /opt/sklep/.sklep.jar.nowy && mv -T /opt/sklep/.sklep.jar.nowy /opt/sklep/sklep.jar
        systemctl restart sklep; zakoncz 1 "wdrozenie $wersja wycofane"
    fi
}
main "\$@"
EOF
sudo chmod 755 /usr/local/bin/wdroz-sklep.sh; sudo bash -n /usr/local/bin/wdroz-sklep.sh && echo "skladnia OK"'

- Restart

Q: Do czego służy TimeoutStopSec=45?

- Opóźnia start o 45 s
- Restart co 45 s
- Limit czasu wdrożenia
E: Musi być dłuższy niż czas graceful shutdown Spring Boot.

- Restartuje po kodzie 0
E: Kod 143 (sukces) nie wywoła restartu.

Q: Co ogranicza MemoryMax=2G?

- Rozmiar dysku
- Liczbę wątków
- Liczbę połączeń
E: Przekroczenie kończy się OOM w obrębie usługi.
#Q
Q: Co robi ProtectSystem=strict?
+ Montuje cały system plików tylko do odczytu dla usługi
- Szyfruje system
- Blokuje sieć
- Ukrywa procesy
E: Zapis możliwy tylko w ReadWritePaths (i /tmp prywatnym).
#Q
Q: Do czego służy ReadWritePaths=?

- Ścieżki w PATH
- Katalogi logów
E: Np. /var/lib/sklep.
#Q
Q: Co robi ProtectHome=yes?
+ Ukrywa katalogi domowe użytkowników przed usługą
- Chroni /home przed usunięciem
- Tworzy katalog domowy
- Montuje /home do zapisu
E: Usługa nie potrzebuje dostępu do /home.
#Q
Q: Co daje PrivateTmp=yes?
+ Usługa ma własny, prywatny /tmp (izolowany od reszty systemu)

- Montuje /tmp tylko do odczytu
- Współdzieli /tmp z innymi

- Ustawia adres IP
