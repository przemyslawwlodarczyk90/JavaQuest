#!/bin/bash
# skrypt-admin.sh - szablon. Uzycie: skrypt-admin.sh [-n] [-h]
set -euo pipefail
readonly NAZWA=${0##*/}
SYMULACJA=0

uzycie() { echo "Uzycie: $NAZWA [-n] [-h]   -n symulacja, -h pomoc"; }
loguj()  {
    local poziom=$1; shift
    logger -t "$NAZWA" -p "user.$poziom" -- "$*" 2>/dev/null || true
    [[ -t 2 ]] && echo "$NAZWA: [$poziom] $*" >&2
    return 0
}
zakoncz() { loguj err "$2"; exit "$1"; }
uruchom() { if (( SYMULACJA )); then echo "[symulacja] $*"; else "$@"; fi; }

while getopts ":nh" opcja; do
    case $opcja in
        n) SYMULACJA=1 ;;
        h) uzycie; exit 0 ;;
        *) uzycie >&2; exit 2 ;;
    esac
done
shift $((OPTIND - 1))

exec 9> "${KATALOG_BLOKAD:-/run/lock}/$NAZWA.lock"
flock -n 9 || { loguj notice "inny egzemplarz juz dziala"; exit 0; }

main() {
    loguj info "start"
    sleep "${CZEKAJ:-0}"
    loguj info "koniec"
}
main "$@"
