P: Napisz skrypt ~/bin/nm-stan.sh, który dla każdego aktywnego połączenia wypisuje nazwę, urządzenie, metodę IPv4, adresy i DNS.
H: nmcli -g NAME,DEVICE c show --active, a potem nmcli -g dla każdego.
S:
cat > ~/bin/nm-stan.sh <<'EOF'
#!/bin/bash
nmcli -g NAME,DEVICE connection show --active | while IFS=: read -r nazwa urz; do
  echo "$nazwa ($urz): $(nmcli -g ipv4.method,ipv4.addresses,ipv4.dns c show "$nazwa" | tr '\n' ' ')"
done
EOF
chmod +x ~/bin/nm-stan.sh
~/bin/nm-stan.sh
