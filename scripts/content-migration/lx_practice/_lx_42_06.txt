    esac

cat > dlugie.sh <<'KONIEC'
#!/bin/bash
set -euo pipefail
args=$(getopt -o nvb: -l dry-run,verbose,bwlimit: -n "${0##*/}" -- "$@") || exit 2
eval set -- "$args"
dry=0; verbose=0; limit=brak
while true; do
    case "$1" in
        -n|--dry-run) dry=1; shift ;;
        -v|--verbose) verbose=1; shift ;;
        -b|--bwlimit) limit=$2; shift 2 ;;
        --) shift; break ;;
    esac
done
echo "dry=$dry verbose=$verbose limit=$limit argumenty: $*"
KONIEC
