P: Zapisz skrypt kopia-katalogu.sh z teorii (opcje -n, -v, -h, --, kontrola liczby argumentów, kod 2 przy złym użyciu, kod 1 przy błędzie rsync) i sprawdź jego składnię.
H: Opcje zbieraj w tablicy opcje, a ścieżki przekazuj do rsync po --.
S:
cat > kopia-katalogu.sh <<'KONIEC'
#!/bin/bash
# kopia-katalogu.sh [-n] [-v] ZRODLO CEL - kopia katalogu rsync-iem
#   -n  tylko symulacja (dry run)    -v  szczegolowy wynik    -h  pomoc
uzycie() { echo "Uzycie: $(basename "$0") [-n] [-v] ZRODLO CEL" >&2; }
opcje=(-a)
while [[ $# -gt 0 ]]; do
    case "$1" in
        -n|--dry-run) opcje+=(-n -i); shift ;;
        -v|--verbose) opcje+=(-v); shift ;;
        -h|--help)    uzycie; exit 0 ;;
        --)           shift; break ;;
        -*)           echo "BLAD: nieznana opcja $1" >&2; uzycie; exit 2 ;;
        *)            break ;;
    esac
done
if [[ $# -ne 2 ]]; then echo "BLAD: potrzebne 2 argumenty, podano $#" >&2; uzycie; exit 2; fi
zrodlo=$1; cel=$2
[[ -d $zrodlo ]] || { echo "BLAD: brak katalogu $zrodlo" >&2; exit 1; }
if rsync "${opcje[@]}" -- "${zrodlo%/}/" "$cel"; then
    echo "kopia OK: $zrodlo -> $cel"
else
    kod=$?; echo "BLAD: rsync zakonczony kodem $kod" >&2; exit 1
fi
KONIEC
chmod +x kopia-katalogu.sh
bash -n kopia-katalogu.sh && echo "skladnia OK"
