import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.lang.management.ManagementFactory;
import java.net.InetSocketAddress;
import java.nio.charset.StandardCharsets;

public class Sklep {
    static final String WERSJA = "@WERSJA@";
    static final boolean ZEPSUTA = @ZEPSUTA@;

    public static void main(String[] args) throws Exception {
        int port = 18080;
        for (String a : args) if (a.startsWith("--server.port=")) port = Integer.parseInt(a.substring(14));
        if (ZEPSUTA || System.getenv("SKLEP_AWARIA") != null) {
            System.err.println("APPLICATION FAILED TO START: brak polaczenia z baza");
            System.exit(1);
        }
        Thread.sleep(Long.parseLong(System.getenv().getOrDefault("SKLEP_START_MS", "1500")));
        HttpServer serwer = HttpServer.create(new InetSocketAddress("127.0.0.1", port), 0);
        serwer.createContext("/actuator/health/readiness", e -> odpowiedz(e, "{\"status\":\"UP\"}"));
        serwer.createContext("/wersja", e -> odpowiedz(e, WERSJA));
        long stopMs = Long.parseLong(System.getenv().getOrDefault("SKLEP_STOP_MS", "0"));
        Runtime.getRuntime().addShutdownHook(new Thread(() -> {
            System.out.println("Commencing graceful shutdown");
            try { Thread.sleep(stopMs); } catch (InterruptedException ignored) { }
            serwer.stop(0);
            System.out.println("Shutdown completed");
        }));
        serwer.start();
        System.out.println("Started Sklep " + WERSJA + " on port " + port
            + " (PID " + ProcessHandle.current().pid() + "), JVM: "
            + ManagementFactory.getRuntimeMXBean().getInputArguments()
            + ", app.nazwa=" + System.getProperty("app.nazwa"));
    }

    static void odpowiedz(HttpExchange e, String tekst) throws IOException {
        byte[] b = tekst.getBytes(StandardCharsets.UTF_8);
        e.sendResponseHeaders(200, b.length);
        e.getResponseBody().write(b);
        e.close();
    }
}
KONIEC
zbuduj() {
    local w=$1 z=$2 k
    k=$(mktemp -d)
    sed -e "s/@WERSJA@/$w/" -e "s/@ZEPSUTA@/$z/" src/Sklep.java > "$k/Sklep.java"
    javac -d "$k/klasy" "$k/Sklep.java" && jar --create --file "app/sklep-$w.jar" --main-class Sklep -C "$k/klasy" .
    rm -rf -- "$k"
    ( cd app && sha256sum "sklep-$w.jar" > "sklep-$w.jar.sha256" )
}
