#EX
P: Pokaż bieżący katalog powłoki.

#EX

lsof -a -p "$$" -d 0,1,2 -Fpfn

P: Otwórz własny plik do zapisu na FD 9, sprawdź go przez lsof i zamknij deskryptor.
H: exec 9> otwiera plik w bieżącej powłoce.

lsof -a -p "$$" -d 9

#EX

#EX

#EX
P: Utwórz własny plik, otwórz go, usuń jego nazwę i sprawdź +L1, po czym zamknij FD.

S:

exec 9>"$plik"

exec 9>&-
#EX

#EX
P: Sprawdź liczbę hardlinków własnego otwartego pliku przed i po usunięciu jednej z dwóch nazw.
H: Dopóki druga nazwa istnieje, liczba dowiązań nie wynosi zero.
S:

printf "dane\n" > "$katalog/jeden"
ln "$katalog/jeden" "$katalog/dwa"
exec 9<"$katalog/jeden"
stat -c "%h" "$katalog/jeden"
rm -- "$katalog/jeden"
stat -c "%h" "$katalog/dwa"
exec 9<&-
rm -- "$katalog/dwa"
rmdir -- "$katalog"
#EX
P: Znajdź otwarte pliki w małym własnym katalogu przez +D i posprzątaj.
H: Nie stosuj rekurencyjnego +D do ogromnych drzew bez potrzeby.
S:
katalog=$(mktemp -d)
printf "dane\n" > "$katalog/proba.txt"
exec 9<"$katalog/proba.txt"
lsof +D "$katalog"
exec 9<&-
rm -- "$katalog/proba.txt"
rmdir -- "$katalog"
#EX
P: Zbadaj cwd procesu uruchomionego we własnym katalogu.
H: Użyj własnego sleep i poczekaj na jego koniec.
S:

(cd "$katalog" && exec sleep 3) &
pid=$!
sleep 0.2
lsof -a -p "$pid" -d cwd
wait "$pid"
rmdir -- "$katalog"
