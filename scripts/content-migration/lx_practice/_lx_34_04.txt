P: Napisz skrypt ~/bin/delta-stat.sh, który dla podanego pliku i celu wypisze procent danych przesłanych dosłownie.

cat > ~/bin/delta-stat.sh <<'EOF'
#!/bin/bash
plik=${1:?plik}; cel=${2:?HOST:katalog}
out=$(rsync -a --stats --no-human-readable "$plik" "$cel")
lit=$(echo "$out" | awk -F': ' '/Literal data/ {gsub(/[^0-9]/,"",$2); print $2}')
mat=$(echo "$out" | awk -F': ' '/Matched data/ {gsub(/[^0-9]/,"",$2); print $2}')
suma=$((lit + mat))
[ "$suma" -gt 0 ] && echo "literalnie: $((100 * lit / suma))% ($lit z $suma B)" || echo 'brak transferu'
EOF

- Tylko przy -z
