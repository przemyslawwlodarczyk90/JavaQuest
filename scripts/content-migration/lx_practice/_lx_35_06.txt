#!/bin/bash

for j in $jails; do
    st=$(sudo fail2ban-client status "$j")
    n=$(echo "$st" | grep -oP 'Currently banned:\s*\K\d+')
    ip=$(echo "$st" | grep -oP 'Banned IP list:\s*\K.*')

done
EOF
