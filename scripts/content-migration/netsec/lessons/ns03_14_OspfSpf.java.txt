import java.util.*;

public class OspfSpf {
    record Edge(String to, int cost) {}

    // Dijkstra ze ścieżkami równego kosztu (ECMP) — dokładnie to, co OSPF liczy na LSDB
    static Map<String, Integer> spf(Map<String, List<Edge>> g, String src, Map<String, Set<String>> prev) {
        Map<String, Integer> dist = new HashMap<>();
        g.keySet().forEach(n -> { dist.put(n, Integer.MAX_VALUE); prev.put(n, new TreeSet<>()); });
        dist.put(src, 0);
        PriorityQueue<Map.Entry<String, Integer>> pq = new PriorityQueue<>(Map.Entry.comparingByValue());
        pq.add(Map.entry(src, 0));
        while (!pq.isEmpty()) {
            var cur = pq.poll();
            if (cur.getValue() > dist.get(cur.getKey())) continue;
            for (Edge e : g.get(cur.getKey())) {
                int nd = cur.getValue() + e.cost();
                if (nd < dist.get(e.to())) { dist.put(e.to(), nd); prev.get(e.to()).clear(); prev.get(e.to()).add(cur.getKey()); pq.add(Map.entry(e.to(), nd)); }
                else if (nd == dist.get(e.to())) prev.get(e.to()).add(cur.getKey());   // ECMP
            }
        }
        return dist;
    }

    static List<String> paths(Map<String, Set<String>> prev, String src, String dst) {
        if (dst.equals(src)) return List.of(src);
        List<String> out = new ArrayList<>();
        for (String p : prev.get(dst)) for (String head : paths(prev, src, p)) out.add(head + ">" + dst);
        return out;
    }

    static int cost(int bwMbps, int refMbps) { return Math.max(1, refMbps / bwMbps); }

    public static void main(String[] args) {
        int ref = args.length > 0 ? Integer.parseInt(args[0]) : 100;
        String[][] links = {{"R1", "R2", "1000"}, {"R2", "R4", "1000"}, {"R1", "R3", "100"}, {"R3", "R4", "100"}, {"R2", "R3", "10"}};
        Map<String, List<Edge>> g = new TreeMap<>();
        for (String[] l : links) {
            int c = cost(Integer.parseInt(l[2]), ref);
            g.computeIfAbsent(l[0], k -> new ArrayList<>()).add(new Edge(l[1], c));
            g.computeIfAbsent(l[1], k -> new ArrayList<>()).add(new Edge(l[0], c));
        }
        Map<String, Set<String>> prev = new HashMap<>();
        Map<String, Integer> dist = spf(g, "R1", prev);
        System.out.println("ref=" + ref + " Mb/s, koszty od R1: " + new TreeMap<>(dist));
        System.out.println("R1 -> R4: " + paths(prev, "R1", "R4"));
    }
}
